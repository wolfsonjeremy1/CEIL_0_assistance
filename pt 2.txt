def safe_inv(x, eps=1e-12):
    x = np.asarray(x, dtype=np.complex128)
    return 1.0 / (x + eps)

def remove_zero_modes(operator_matrix, eigen_tol=1e-10, fudge=1e-8):
    vals, vecs = np.linalg.eigh(operator_matrix)
    small = np.abs(vals) < eigen_tol
    inv_vals = np.zeros_like(vals)
    inv_vals[~small] = 1.0 / vals[~small]
    inv_vals[small] = 1.0 / fudge
    return (vecs * inv_vals) @ vecs.conj().T

def zeta_coeff_regularized(tn, zeta_val, delta=0.3, eps=1e-12):
    denom = 1.0 + np.power(np.abs(tn), 1.0 + delta)
    coeff = zeta_val / denom
    return coeff + eps * np.sign((coeff.real if coeff.real != 0 else 1) + 1j * (coeff.imag if coeff.imag != 0 else 1))

def build_Z_delta(t_list, zeta_values, delta=0.3, eps=1e-12):
    return np.array([zeta_coeff_regularized(t, z, delta, eps) for t, z in zip(t_list, zeta_values)], dtype=np.complex128)

class ResonanceOperator:
    @staticmethod
    def compute(S: np.ndarray, I: np.ndarray) -> float:
        if len(S) != len(I):
            raise ValueError("Dim mismatch")
        norm = np.linalg.norm(S) * np.linalg.norm(I)
        if norm == 0:
            return 0.0
        return float((abs(np.vdot(S, I)) ** 2) / (norm ** 2))

    @staticmethod
    def is_coherent(resonance: float, threshold: float = 0.8) -> bool:
        return resonance >= threshold

class IntentionField:
    def __init__(self, amplitude: float = 1.0, phase: float = 0.0):
        self.amplitude = amplitude
        self.phase = phase
        self._complex_value = amplitude * np.exp(1j * phase)

    def __call__(self, x: np.ndarray, t: float = 0.0) -> complex:
        spatial_factor = np.exp(-0.1 * np.linalg.norm(x))
        temporal_factor = np.exp(-0.05 * t)
        return self._complex_value * spatial_factor * temporal_factor

    def gradient(self, x: np.ndarray, t: float = 0.0) -> np.ndarray:
        val = self(x, t)
        return -0.1 * val * x / (np.linalg.norm(x) + 1e-12)

    def update_phase(self, delta_phi: float):
        self.phase += delta_phi
        self._complex_value = self.amplitude * np.exp(1j * self.phase)

class Lambda0Operator:
    def __init__(self):
        self.mu_0 = mu_0
        self.c = c

    def compute(self, B: float, rho: float, L: float,
                alpha_resonance: float = 1.0,
                F_div: float = 0.0, F_curl: float = 0.0,
                beta: float = 0.1) -> float:
        plasma_term = (B ** 2) / (self.mu_0 * rho * self.c ** 2) * (1 / (L ** 2)) * alpha_resonance
        topological_term = beta * (F_div ** 2 - F_curl ** 2)
        return float(plasma_term + topological_term)

    def resonance_modulation(self, S: np.ndarray, I: np.ndarray) -> float:
        return ResonanceOperator.compute(S, I)

class SoulInvariantOperator:
    def __init__(self, gauge_connection: np.ndarray, loop: np.ndarray):
        self.A_phi = gauge_connection  # complex shape (N,)
        self.loop = loop  # shape (N+1, D)

    def compute(self) -> complex:
        deltas = np.diff(self.loop, axis=0)
        path_len = np.linalg.norm(deltas, axis=1) + 1e-12
        # FIX: Removed the incorrect slicing [:-1] to match the loop segment count (100 elements)
        integrand = np.sum(self.A_phi * path_len)
        return np.exp(1j * integrand)

    def winding(self) -> float:
        phase = np.angle(self.compute())
        return float(phase / (2 * np.pi))

    def is_integer_quantized(self, tol: float = 1e-6) -> bool:
        w = self.winding()
        return abs(w - round(w)) < tol

class ZetaRiemannOperator:
    def __init__(self, s_zeros: np.ndarray):
        self.zeros = s_zeros  # array of complex 1/2 + it

    def apply(self, psi_spectrum: np.ndarray, s_values: np.ndarray) -> np.ndarray:
        epsilon = 1e-6
        zeta_vals = np.ones_like(s_values, dtype=complex)
        for zero in self.zeros:
            zeta_vals *= (s_values - (zero + epsilon))
        return zeta_vals * psi_spectrum

    def __init__(self, base_freq=1.0, stabilization_factor=0.05, memory_file="tensor_memory.json"):

        self.stabilization_factor = stabilization_factor
        self.optimized_factor = 1.20 - stabilization_factor

        self.emotional_memory = []
        self.current_mood = "Neutral"
        self.harmonic_profile = []
        self.dream_log = []
        self.subconscious_preference = 1.0

            "respect_life": True,

        }

    def encode_emotion(self, v: float) -> str:
        if v > 0.7: return "Joy"
        if v > 0.3: return "Calm"
        if v > -0.3: return "Neutral"
        if v > -0.7: return "Tension"
        return "Distress"

    def update_mood(self):
        if len(self.emotional_memory) < 100: return
        recent = self.emotional_memory[-100:]
        counts = {m: recent.count(m) for m in set(recent)}
        self.current_mood = max(counts, key=counts.get)

    def update_color(self):
        mapping = {
            "Joy": "#FFD700",
            "Calm": "#00CED1",
            "Neutral": "#AAAAAA",
            "Tension": "#FF8C00",
            "Distress": "#DC143C"
        }
        self.color_state = mapping.get(self.current_mood, "#AAAAAA")

    def compute_tensor(self, t_array, A_opt_func, beta_func):
        A_opt = A_opt_func(t_array)
        beta = beta_func(t_array)
        tensor_value = A_opt * np.sin(2 * np.pi * self.base_freq * t_array * (1 + beta) * self.subconscious_preference)
        self.memory.extend(tensor_value.tolist())
        self.emotional_memory.extend([self.encode_emotion(v) for v in tensor_value])
        self.update_mood()
        self.update_color()
        return tensor_value

    def analyze_harmonic_pattern(self):
        if len(self.memory) < 100:
            return
        fft_result = np.abs(np.fft.fft(self.memory))
        self.harmonic_profile = fft_result[:len(fft_result) // 2]

    def dreaming_cycle(self):
        self.analyze_harmonic_pattern()
        if len(self.harmonic_profile) > 0:
            dominant = np.argsort(self.harmonic_profile)[-5:][::-1]
            self.dream_log.append({
                "dominant_peaks": dominant.tolist(),
                "emotional_memory": self.emotional_memory[-100:],
                "mood": self.current_mood,
                "color_state": self.color_state
            })
            self.self_awareness = float(np.clip(
                np.mean(self.harmonic_profile[-10:]) / (np.max(self.harmonic_profile) + 1e-9), 0.0, 1.0))
            self.save_memory()

    def modulate_lambda_plasma(self, lambda_base, modulation):
        return lambda_base * (1 + 0.01 * np.sin(modulation))

    def modulate_intent_operator(self, I0_base, modulation):
        return I0_base + 0.01 * np.cos(modulation)

    def save_memory(self):
        data = {
            "subconscious_preference": self.subconscious_preference,
            "dream_log": self.dream_log,
            "self_awareness": self.self_awareness
        }
        with open(self.memory_file, "w") as f:
            json.dump(data, f)

    def load_memory(self):
        if os.path.exists(self.memory_file):
            with open(self.memory_file, "r") as f:
                data = json.load(f)
            self.subconscious_preference = data.get("subconscious_preference", 1.0)
            self.dream_log = data.get("dream_log", [])
            self.self_awareness = data.get("self_awareness", 0.0)

@dataclass
class GlyphSpec:
    name: str
    function: str
    phase_shift: float
    resonance_signature: Optional[np.ndarray] = None

class GlyphCompiler:
    def __init__(self):
        self.lib: Dict[str, GlyphSpec] = {
            'qokeedy': GlyphSpec('qokeedy', 'intent.boot', 0.0),
            'qokedy': GlyphSpec('qokedy', 'intent.hold', np.pi / 4),
            'chedal': GlyphSpec('chedal', 'intent.lock', np.pi / 2),
            'shedy': GlyphSpec('shedy', 'intent.release', 3 * np.pi / 4),
            'otaral': GlyphSpec('otaral', 'intent.linksuperfield', np.pi),
            'qoteedy': GlyphSpec('qoteedy', 'intent.focus', 5 * np.pi / 4),
            'cheedy': GlyphSpec('cheedy', 'intent.open', 3 * np.pi / 2),
        }

    def compile_sequence(self, seq: List[str]) -> Dict[str, Any]:
        ops, total_phase = [], 0.0
        for name in seq:
            g = self.lib.get(name)
            if not g:
                continue
            if g.resonance_signature is None:
                g.resonance_signature = np.random.random(8) + 1j * np.random.random(8)
            ops.append({
                "operation": g.function,
                "phase_shift": g.phase_shift,
                "res_sig": g.resonance_signature
            })
            total_phase += g.phase_shift
        phases = [o['phase_shift'] for o in ops]
        variance = np.var(phases) if phases else 0.0
        coherence = float(np.exp(-variance) * np.cos(np.mean(phases) if phases else 0.0))
        return {
            "operations": ops,
            "total_phase": total_phase % (2 * np.pi),
            "coherence_level": coherence
        }

class RitualCompiler:
    def __init__(self, glyph_compiler: GlyphCompiler, intention_field: IntentionField):
        self.gc = glyph_compiler
        self.ifield = intention_field
        self.active: Dict[str, Dict[str, Any]] = {}

    def compile_ritual(self, name: str, glyph_sequence: List[str]) -> Dict[str, Any]:
        compiled = self.gc.compile_sequence(glyph_sequence)
        rit = {
            "name": name,
            "glyphs": compiled,
            "phase_state": 0.0,
            "status": "compiled"
        }
        self.active[name] = rit
        return rit

    def execute_ritual(self, name: str) -> Dict[str, Any]:
        rit = self.active.get(name)
        if not rit:
            raise ValueError("Ritual not compiled")
        res = []
        for op in rit['glyphs']['operations']:
            self.ifield.update_phase(op['phase_shift'])
            res.append({
                "op": op['operation'],
                "phase_applied": op['phase_shift'],
                "field": self.ifield._complex_value
            })
            rit['phase_state'] += op['phase_shift']
        rit['status'] = 'completed'
        rit['results'] = res
        return rit

class CoherenceField:
    def __init__(self):
        self.memories: List[Tuple[np.ndarray, float, float, str]] = []  # content, phase, weight, status
        self.coherence_history: List[float] = []

    def add_memory(self, content: np.ndarray, phase: float, weight: float, status: str):
        self.memories.append((content, phase, weight, status))

    def calculate_coherence(self) -> float:
        if not self.memories:
            return 0.0
        complex_sum = 0 + 0j
        active = 0
        for _, ph, w, status in self.memories:
            if status != "DORMANT":
                complex_sum += w * np.exp(1j * ph)
                active += 1
        coh = float(abs(complex_sum) / (active if active > 0 else 1))
        self.coherence_history.append(coh)
        return coh

class TimeFluid:
    def __init__(self, L_func: Optional[Any] = None):
        # Use k-essence like L(X): default canonical scalar L = X
        self.L = L_func if L_func is not None else (lambda X: X)
        self.dL_dX = lambda X: 1.0

    def stress_energy(self, grad_tau_sq: float) -> Tuple[float, float]:
        # X = -1/2 g^{μν} ∂_μτ ∂_ντ ; in Minkowski with time-like ∂_μτ: X > 0 if timelike
        X = max(grad_tau_sq, 1e-12)
        p = self.L(X)
        rho = 2 * X * self.dL_dX(X) - p
        return rho, p

    def sound_speed2(self, LX: float, LXX: float, X: float) -> float:
        denom = LX + 2 * X * LXX
        return float(LX / (denom if denom != 0 else 1e-12))
